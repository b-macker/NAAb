--- NAAb syntax reference ---
// Imports (required at top of file, before main)
use agent
use env
use string
use json
use time

// Everything runs inside main { }
main {
    // Variables
    let x = 42
    let name = "hello"

    // String concatenation with +
    let msg = "count: " + string(x)

    // Print to stdout
    print("hello world")

    // Command-line args (returns array, index 0 is first arg)
    let args = env.get_args()
    let prompt = args[0]
    // Join multiple args:
    let i = 1
    while i < args.length() {
        prompt = prompt + " " + args[i]
        i = i + 1
    }

    // NO for loops — use while
    let j = 0
    while j < 3 {
        print("iter " + string(j))
        j = j + 1
    }

    // Agent API
    let handle = agent.create("agent_name")  // name must match govern.json agents key
    let resp = agent.send(handle, "your message here")
    let content = resp.get("content")  // response text is in .get("content")

    // Null coalescing
    let val = resp.get("missing_key") ?? "default"

    // Dicts
    let d = {"key": "value", "num": 42}
    let v = d.get("key")

    // Functions (defined INSIDE main is ok)
    fn helper(x) {
        return x + 1
    }

    // Error handling
    try {
        let r = agent.send(handle, "msg")
        return r
    } catch (e) {
        print("error: " + string(e))
        return {"error": string(e)}
    }

    // Agent environment/coherence
    let env_data = agent.environment(handle)
    let state = env_data.get("state")
    let coherence = state.get("coherence") ?? -1
    let turns = state.get("turns_used") ?? 0
}
